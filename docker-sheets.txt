 Maintenant copie ce script sur le VPS et rends-le exécutable :                                                                                                               

  # Sur le VPS — une seule fois                                                                                                                                                
  cp /var/www/ritoto/update.sh ~/update.sh                                                                                                                                     
  chmod +x ~/update.sh                                                                                                                                                         
                                                                                                                                                                               
  ---                                                                                                                                                                          
  Utilisation du script                                                                                                                                                        
                                                                                                                                                                             
  bash ~/update.sh           # mise a jour complete (pull + build + migrate)
  bash ~/update.sh nginx     # apres avoir modifie nginx.conf                                                                                                                  
  bash ~/update.sh front     # apres avoir modifie du code React                                                                                                               
  bash ~/update.sh back      # apres avoir modifie du code Django/Python                                                                                                       
  bash ~/update.sh restart   # redemarrer tout sans rebuild                                                                                                                    
                                                                                                                                                                               
  ---
  Commandes Django sur le serveur                                                                                                                                              
                                                                                                                                                                               
  Toutes les commandes Django s'exécutent via docker compose exec web :
                                                                                                                                                                               
  COMPOSE="docker compose -f /var/www/ritoto/docker-compose.prod.yml"                                                                                                        
                                                                                                                                                                               
  # Migrations                                                                                                                                                               
  $COMPOSE exec web python manage.py migrate --noinput
                                                                                                                                                                               
  # Creer un superutilisateur
  $COMPOSE exec web python manage.py createsuperuser                                                                                                                           
                                                                                                                                                                             
  # Shell Django interactif                                                                                                                                                    
  $COMPOSE exec web python manage.py shell
                                                                                                                                                                               
  # Collectstatic (fichiers admin)                                                                                                                                           
  $COMPOSE exec web python manage.py collectstatic --noinput
                                                                                                                                                                               
  # Voir les migrations en attente
  $COMPOSE exec web python manage.py showmigrations                                                                                                                            
                                                                                                                                                                             
  # Creer une migration apres modif de models.py                                                                                                                               
  $COMPOSE exec web python manage.py makemigrations
                                                                                                                                                                               
  # Logs en direct                                                                                                                                                             
  $COMPOSE logs -f web
  $COMPOSE logs -f nginx
