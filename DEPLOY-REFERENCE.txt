FROM python:3.11-slim

ENV PYTHONDONTWRITEBYTECODE=1
ENV PYTHONUNBUFFERED=1
ENV DJANGO_SETTINGS_MODULE=config.settings.production

WORKDIR /app

# Dependances systeme
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential libpq-dev \
    && rm -rf /var/lib/apt/lists/*

# Dependances Python
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Code source
COPY . .

# collectstatic n'a pas besoin de la vraie SECRET_KEY ni de la DB
RUN DJANGO_SECRET_KEY=build-placeholder-key \
    python manage.py collectstatic --noinput

EXPOSE 8000 8001

services:

  # Base de donnees
  postgres:
    image: postgres:16-alpine
    container_name: ritoto-postgres
    env_file: .env.production
    volumes:
      - postgres_data:/var/lib/postgresql/data
    networks:
      - internal
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U $$POSTGRES_USER"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: always

  # Cache & Channel Layer
  redis:
    image: redis:7-alpine
    container_name: ritoto-redis
    command: redis-server --appendonly yes --maxmemory 256mb --maxmemory-policy allkeys-lru
    volumes:
      - redis_data:/data
    networks:
      - internal
    healthcheck:
      test: ["CMD", "redis-cli", "ping"]
      interval: 10s
      timeout: 3s
      retries: 5
    restart: always

  # Backend HTTP (Gunicorn)
  backend:
    build:
      context: .
      dockerfile: Dockerfile.prod
    container_name: ritoto-backend
    command: gunicorn config.wsgi:application --bind 0.0.0.0:8000 --workers 3 --timeout 120
    env_file: .env.production
    volumes:
      - static_data:/app/static
      - media_data:/app/media
    networks:
      - internal
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_healthy
    restart: always

  # Backend WebSockets (Daphne)

    build:
      context: .

      - media_data:/app/media
    networks:
      - internal

        condition: service_healthy

# HTTP -> HTTPS redirect
server {
    listen 80;
    server_name api.ritoto-campus.com;

    location /.well-known/acme-challenge/ {
        root /var/www/certbot;
    }

    location / {
        return 301 https://$host$request_uri;
    }
}

# HTTPS - API
server {
    listen 443 ssl;
    server_name api.ritoto-campus.com;

    ssl_certificate     /etc/letsencrypt/live/api.ritoto-campus.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/api.ritoto-campus.com/privkey.pem;
    ssl_protocols       TLSv1.2 TLSv1.3;
    ssl_ciphers         HIGH:!aNULL:!MD5;
    ssl_session_cache   shared:SSL:10m;

    # WebSockets (Daphne)
    location /ws/ {
        proxy_pass http://daphne:8001;
        proxy_http_version 1.1;
        proxy_set_header Upgrade    $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host       $host;
        proxy_set_header X-Real-IP  $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_read_timeout 86400;
        proxy_send_timeout 86400;
    }

    # API REST (Gunicorn)
    location /api/ {
        proxy_pass http://backend:8000;
        proxy_set_header Host               $host;
        proxy_set_header X-Real-IP          $remote_addr;
        proxy_set_header X-Forwarded-For    $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto  $scheme;
        proxy_connect_timeout 60s;
        proxy_read_timeout    120s;
    }

    # Admin Django
    location /admin/ {
        proxy_pass http://backend:8000;
        proxy_set_header Host               $host;
        proxy_set_header X-Real-IP          $remote_addr;
        proxy_set_header X-Forwarded-For    $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto  $scheme;
    }

    # Fichiers statiques Django (collectstatic)
    location /static/ {
        alias /var/www/static/;
        expires 30d;
        add_header Cache-Control "public";
    }

    # Fichiers media (uploads)
    location /media/ {
        alias /var/www/media/;
        expires 7d;
        add_header Cache-Control "public";
    }
}

#!/bin/bash
# Planifier : 0 2 * * * /var/www/ritoto/docker/scripts/backup.sh >> /var/log/ritoto-backup.log 2>&1

set -euo pipefail

COMPOSE_FILE="/var/www/ritoto/docker-compose.prod.yml"
BACKUP_DIR="/var/www/ritoto/backups"
DATE=$(date +%Y%m%d-%H%M)
KEEP=14

mkdir -p "$BACKUP_DIR"

echo "[$(date '+%Y-%m-%d %H:%M:%S')] === Debut backup ==="

# 1. Dump PostgreSQL
docker compose -f "$COMPOSE_FILE" exec -T postgres \
    pg_dump -U "${POSTGRES_USER:-ritoto}" "${POSTGRES_DB:-ritoto}" \
    > "$BACKUP_DIR/db-$DATE.sql"

# 2. Archive media
tar -czf "$BACKUP_DIR/media-$DATE.tar.gz" \
    -C /var/www/ritoto media/ 2>/dev/null || true

# 3. Archive globale
tar -czf "$BACKUP_DIR/ritoto-backup-$DATE.tar.gz" \
    "$BACKUP_DIR/db-$DATE.sql" \
    "$BACKUP_DIR/media-$DATE.tar.gz"

# 4. Nettoyer les fichiers intermediaires
rm -f "$BACKUP_DIR/db-$DATE.sql" "$BACKUP_DIR/media-$DATE.tar.gz"

# 5. Rotation : garder seulement les $KEEP derniers backups
ls -t "$BACKUP_DIR"/ritoto-backup-*.tar.gz 2>/dev/null | tail -n +$((KEEP + 1)) | xargs -r rm -f

SIZE=$(du -sh "$BACKUP_DIR/ritoto-backup-$DATE.tar.gz" 2>/dev/null | cut -f1)
echo "[$(date '+%Y-%m-%d %H:%M:%S')] === Backup termine : ritoto-backup-$DATE.tar.gz ($SIZE) ==="

#!/bin/bash
# Usage : ./restore.sh /var/www/ritoto/backups/ritoto-backup-20260322-0200.tar.gz

set -euo pipefail

COMPOSE_FILE="/var/www/ritoto/docker-compose.prod.yml"
BACKUP_FILE="${1:-}"

if [[ -z "$BACKUP_FILE" || ! -f "$BACKUP_FILE" ]]; then
    echo "Usage: $0 <chemin-du-backup.tar.gz>"
    exit 1
fi

echo "[$(date '+%Y-%m-%d %H:%M:%S')] Restauration depuis : $BACKUP_FILE"

WORK_DIR=$(mktemp -d)
trap "rm -rf $WORK_DIR" EXIT

# Extraire le backup
tar -xzf "$BACKUP_FILE" -C "$WORK_DIR"

# 1. Arreter les services applicatifs (garder postgres)
docker compose -f "$COMPOSE_FILE" stop backend daphne nginx frontend 2>/dev/null || true

# 2. Restaurer la base PostgreSQL
DB_FILE=$(find "$WORK_DIR" -name "db-*.sql" | head -1)
if [[ -f "$DB_FILE" ]]; then
    docker compose -f "$COMPOSE_FILE" exec -T postgres \
        psql -U "${POSTGRES_USER:-ritoto}" "${POSTGRES_DB:-ritoto}" < "$DB_FILE"
fi

# 3. Restaurer les media
MEDIA_FILE=$(find "$WORK_DIR" -name "media-*.tar.gz" | head -1)
if [[ -f "$MEDIA_FILE" ]]; then
    tar -xzf "$MEDIA_FILE" -C /var/www/ritoto/
fi

# 4. Redemarrer tout
docker compose -f "$COMPOSE_FILE" up -d

echo "[$(date '+%Y-%m-%d %H:%M:%S')] === Restauration terminee ==="
